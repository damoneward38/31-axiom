// src/components/ParticleField.tsx

import React, { useRef, useEffect, useCallback } from "react";
import { Particle, GravityWell } from "../types/quantum.types";
import { renderParticle, renderEntanglementLink } from "../utils/quantumEngine";

interface ParticleFieldProps {
  particles:    Particle[];
  gravityWells: GravityWell[];
  onCanvasClick:(x: number, y: number) => void;
  width:        number;
  height:       number;
}

export const ParticleField: React.FC<ParticleFieldProps> = ({
  particles,
  gravityWells,
  onCanvasClick,
  width,
  height,
}) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  // ─── Render Frame ─────────────────────────────────────────────
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext("2d", { alpha: false });
    if (!ctx) return;

    // Clear with dark background + subtle trail effect
    ctx.fillStyle = "rgba(5, 5, 15, 0.85)";
    ctx.fillRect(0, 0, width, height);

    // Draw grid overlay
    drawGrid(ctx, width, height);

    // Draw entanglement links (behind particles)
    const entangled = particles.filter(p => p.state === "entangled");
    for (let i = 0; i < entangled.length; i++) {
      for (let j = i + 1; j < entangled.length; j++) {
        renderEntanglementLink(ctx, entangled[i], entangled[j]);
      }
    }

    // Draw gravity wells
    for (const well of gravityWells) {
      drawGravityWell(ctx, well);
    }

    // Draw particles
    for (const particle of particles) {
      renderParticle(ctx, particle);
    }
  }, [particles, gravityWells, width, height]);

  // ─── Grid Drawing ─────────────────────────────────────────────
  const drawGrid = (ctx: CanvasRenderingContext2D, w: number, h: number) => {
    ctx.save();
    ctx.strokeStyle = "rgba(0, 245, 255, 0.03)";
    ctx.lineWidth   = 0.5;

    const step = 60;
    for (let x = 0; x < w; x += step) {
      ctx.beginPath();
      ctx.moveTo(x, 0);
      ctx.lineTo(x, h);
      ctx.stroke();
    }
    for (let y = 0; y < h; y += step) {
      ctx.beginPath();
      ctx.moveTo(0, y);
      ctx.lineTo(w, y);
      ctx.stroke();
    }
    ctx.restore();
  };

  // ─── Gravity Well Visual ──────────────────────────────────────
  const drawGravityWell = (ctx: CanvasRenderingContext2D, well: GravityWell) => {
    ctx.save();

    // Outer ring
    const gradient = ctx.createRadialGradient(
      well.x, well.y, 0,
      well.x, well.y, well.radius
    );
    gradient.addColorStop(0,   "rgba(191, 0, 255, 0.2)");
    gradient.addColorStop(0.5, "rgba(191, 0, 255, 0.05)");
    gradient.addColorStop(1,   "rgba(191, 0, 255, 0)");

    ctx.beginPath();
    ctx.arc(well.x, well.y, well.radius, 0, Math.PI * 2);
    ctx.fillStyle   = gradient;
    ctx.fill();

    // Core dot
    ctx.beginPath();
    ctx.arc(well.x, well.y, 4, 0, Math.PI * 2);
    ctx.fillStyle   = "#bf00ff";
    ctx.shadowBlur  = 20;
    ctx.shadowColor = "#bf00ff";
    ctx.fill();

    ctx.restore();
  };

  // ─── Click Handler ────────────────────────────────────────────
  const handleClick = useCallback(
    (e: React.MouseEvent<HTMLCanvasElement>) => {
      const rect = canvasRef.current?.getBoundingClientRect();
      if (!rect) return;
      onCanvasClick(e.clientX - rect.left, e.clientY - rect.top);
    },
    [onCanvasClick]
  );

  return (
    <canvas
      ref       ={canvasRef}
      width     ={width}
      height    ={height}
      onClick   ={handleClick}
      className ="quantum-canvas"
      style     ={{
        cursor: "crosshair",
        display: "block",
        borderRadius: "12px",
      }}
      aria-label="Quantum particle simulation field"
    />
  );
};
